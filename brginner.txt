#python calculator

operator = input("Enter an operator(+ - * /): "
num1 = float(input("Enter the First Number: "))
num2 = float(input("Enter the Second Number: "))

if operator == "+":
    result = num1 + num2
    print(result)
elif operator == "-":
    result = num1 - num2
    print(result)
elif operator == "*":
    result = num1 * num2
    print(result)
elif operator == "/":
    result = num1 / num2
    print(result)
else:
    print(f"Operator '{operator}' not a valid operator")




#weight conversion program

weight = float(input("Enter your weight: "))
unit = input("Enter unit (K for Kilogram; L for Pounds: ")

if unit == "K":
    weight = weight * 2.205
    unit = "lbs"
    print(f"Your weight is: {round(weight, 1)} {unit}")
elif unit == "L":
    weight = weight / 2.205
    unit = "Kg"
    print(f"Your weight is: {round(weight, 1)} {unit}")
else:
    print(f"{unit} is not a valid unit")
